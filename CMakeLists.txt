
# kameorepowered - ReXGlue Recompiled Project
#
# SDK boilerplate lives in generated/rexglue.cmake.

cmake_minimum_required(VERSION 3.25)
project(kameorepowered LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Title-update builds: the hand-written hooks call game functions by hardcoded
# vanilla addresses that the update relocates. Until those are re-derived for the
# patched image, -DKAMEO_TU=ON compiles the affected calls out (the custom hooks
# are also absent from kameorepowered_tu_config.toml).
#
# The two variants recompile to different guest code, so they use separate
# generated trees (see out_directory_path in each manifest) and should use
# separate build directories. Declared before including the SDK boilerplate so
# the generated-source selection below can key off it.
option(KAMEO_TU "Build for a baked-in title update (disables vanilla-address hooks)" OFF)

include(generated/rexglue.cmake)

# rexglue.cmake only knows about the entrypoint's default tree. Repoint it at
# the TU tree for -DKAMEO_TU=ON; main.cpp includes the init header by bare name
# so it resolves out of whichever tree is on the include path.
if(KAMEO_TU)
    set(_kameo_generated "${CMAKE_CURRENT_SOURCE_DIR}/generated/tu")
    if(NOT EXISTS "${_kameo_generated}/sources.cmake")
        message(FATAL_ERROR
            "KAMEO_TU=ON but ${_kameo_generated}/sources.cmake is missing.\n"
            "Stage the update and run TU codegen first:\n"
            "  python scripts/extract_tu.py <TU_package> --base assets/default.xex "
            "--output assets/default.xexp --update-dir assets/TU\n"
            "  rexglue codegen kameorepowered_tu_manifest.toml")
    endif()
    include("${_kameo_generated}/sources.cmake")
    set(REXGLUE_ENTRYPOINT_GENERATED_SOURCES ${GENERATED_SOURCES})
    set(REXGLUE_ENTRYPOINT_INCLUDE_DIR "${_kameo_generated}")
elseif(NOT REXGLUE_ENTRYPOINT_GENERATED_SOURCES)
    message(FATAL_ERROR
        "No generated sources. Run codegen first:\n"
        "  rexglue codegen kameorepowered_manifest.toml")
endif()

# Sources. The recompiled guest code is added by rexglue_setup_target from the
# generated tree selected above -- do not list GENERATED_SOURCES here.
set(KAMEOREPOWERED_SOURCES
    src/main.cpp
    src/kameorepowered_hooks_combat.cpp
    src/kameorepowered_hooks_audio.cpp
    src/kameorepowered_hooks_dlc.cpp
    src/kameorepowered_hooks_splitscreen.cpp
)

# Native renderer (work in progress). OFF keeps the shipping build byte-for-byte
# what it was: nothing below is compiled and the xenos plugin stays in charge.
#
# plume lives in thirdparty/ as a plain clone (gitignored, not a submodule yet),
# so it has to be fetched before enabling this. See src/gfx/SHADERS.md.
option(KAMEO_NATIVE_RENDERER "Build the plume-backed native renderer" OFF)

if(KAMEO_NATIVE_RENDERER)
    set(_kameo_plume "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/plume")
    if(NOT EXISTS "${_kameo_plume}/CMakeLists.txt")
        message(FATAL_ERROR
            "KAMEO_NATIVE_RENDERER=ON but ${_kameo_plume} is missing.\n"
            "  git clone --recurse-submodules https://github.com/renderbag/plume.git thirdparty/plume")
    endif()
    add_subdirectory("${_kameo_plume}" plume EXCLUDE_FROM_ALL)
    list(APPEND KAMEOREPOWERED_SOURCES
        src/gfx/kameo_graphics_system.cpp
        src/gfx/kameo_gfx_hooks.cpp
        src/gfx/kameo_draw_path.cpp
        # Generated by scripts/build_shader_cache.py -- 134 unique shaders.
        # Regenerating needs a dumped guest image, see src/gfx/SHADERS.md.
        src/gfx/shader_cache.cpp
    )
    # xxHash is pulled in header-only (XXH_INLINE_ALL) rather than as
    # xxhash.c: this project never enables the C language, so adding a .c source
    # fails the CMake generate step with a missing CMAKE_C_COMPILE_OBJECT.
    # The cache key must be XXH3_64bits over the authored container, matching
    # what XenosRecomp computed when the cache was generated.
endif()

if(WIN32)
    add_executable(kameorepowered WIN32 ${KAMEOREPOWERED_SOURCES} icon/app.rc)
else()
    add_executable(kameorepowered ${KAMEOREPOWERED_SOURCES})
endif()

# GPU_PLUGINS stages rexgpu-xenos next to the executable; it is runtime-loaded
# (--gpu_plugin xenos), so it is never picked up by TARGET_RUNTIME_DLLS.
rexglue_setup_target(kameorepowered GPU_PLUGINS xenos)

if(KAMEO_TU)
    target_compile_definitions(kameorepowered PRIVATE KAMEO_TU=1)
endif()

if(KAMEO_NATIVE_RENDERER)
    # plume is a static library, so its D3D12 imports have to be satisfied here
    # (D3D12CreateDevice, D3D12SerializeRootSignature). dxgi/dxguid already come
    # in via the SDK. Vulkan needs nothing: plume loads it through volk.
    target_link_libraries(kameorepowered PRIVATE plume)
    if(WIN32)
        target_link_libraries(kameorepowered PRIVATE d3d12)
    endif()
    target_include_directories(kameorepowered PRIVATE
        src/gfx
        thirdparty/XenosRecomp/thirdparty/xxHash
    )
    target_compile_definitions(kameorepowered PRIVATE KAMEO_NATIVE_RENDERER=1)
endif()
